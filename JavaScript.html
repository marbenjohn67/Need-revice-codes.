Authentication will be connected to the backend.", "success" ); }); // ========================================== // MESSAGE FUNCTION // ========================================== function showMessage(message, type) { loginMessage.textContent = message; loginMessage.classList.remove( "hidden", "bg-red-100", "text-red-700", "bg-green-100", "text-green-700" ); if (type === "success") { loginMessage.classList.add( "bg-green-100", "text-green-700" ); } else { loginMessage.classList.add( "bg-red-100", "text-red-700" ); } } // ========================================== // FORGOT PASSWORD // ========================================== document .getElementById("forgotPassword") .addEventListener("click", function () { alert( "Please contact your Barangay Administrator to reset your password." ); }); // ========================================== // CREATE ACCOUNT // ========================================== document .getElementById("registerButton") .addEventListener("click", function () { alert( "Registration page will be added in the next version." ); });
